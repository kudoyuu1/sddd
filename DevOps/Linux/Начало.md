Команды linux:
```
whoiam #отображает какой я пользователь
id #отображает всю информацию про пользователя, в каких группах состоит и какие права
pwd #отображает в каком катологе я нахожусь
echo #Вывод какого либо сообщения на терминале
ls #содержимое католога
cd #пермешение по катологам ".." - две точки позволяют спустится до родительского каталога "-" на одну ниже
touch #команда созадающая пустой файл
mkdir #создает директории
cp #создание копии файла
mv #переименовывает файлы и перемешает файлы по катологам
rm #удаление файла
rmdir #удаление каталога
file #показывает иформацию о файле
cat #позволяет просматрвать содержимое файла
less #Команда позволяющая просматривать файл, без полной его загрузки в память, позволяет также искать в файле нужную информацию
history #показывает всю историю вводимых команд
find #команда для поиска "-name" поиск по имени "-tupe" поиск по типу
man # команда чтобы получить справочную информацию по командам
whatis # команда дает краткую сводку по другим командам
alias #команда чтобы задать комманаду для последоватиельности и или длиной команде, чтобы ее не повторять
exit # выход их оболочки
head #показывает начало файла
tail #Противоположность head
chown #меняет владельца и группу файла
chmod #меняет права доступа файла
```
